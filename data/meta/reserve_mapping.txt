// ── Raster helpers ─────────────────────────────────────────────────────────────

function hexToRgb(hex: string): [number, number, number] {
  const r = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
  return r ? [parseInt(r[1], 16), parseInt(r[2], 16), parseInt(r[3], 16)] : [0, 0, 0];
}

function rampColor(pct: number, ramp: string[]): string {
  const n = ramp.length;
  const i = Math.min(n - 2, Math.floor(pct * (n - 1)));
  const t = pct * (n - 1) - i;
  const c1 = hexToRgb(ramp[i]), c2 = hexToRgb(ramp[i + 1]);
  return `rgb(${Math.round(c1[0] + t * (c2[0] - c1[0]))},${Math.round(c1[1] + t * (c2[1] - c1[1]))},${Math.round(c1[2] + t * (c2[2] - c1[2]))})`;
}

function generateRaster(layerId: string, cols = 26, rows = 18): number[][] {
  const grid: number[][] = [];
  for (let r = 0; r < rows; r++) {
    const row: number[] = [];
    for (let c = 0; c < cols; c++) {
      const cx = c / cols, cy = r / rows;
      const isZ1 = Math.sqrt((cx - 0.60) ** 2 + (cy - 0.35) ** 2) < 0.11;
      const isZ2 = Math.sqrt((cx - 0.33) ** 2 + (cy - 0.58) ** 2) < 0.09;
      const noise = (Math.random() - 0.5) * 0.22;
      let v = 0.15 + noise * 0.3;
      if (layerId === 'prospec') v = isZ1 ? 0.84 + noise * 0.1 : isZ2 ? 0.73 + noise * 0.12 : 0.18 + noise * 0.28;
      else if (layerId === 'aster') v = isZ1 ? 0.77 + noise * 0.12 : isZ2 ? 0.63 + noise * 0.14 : 0.32 + noise * 0.33;
      else if (layerId === 'ndvi') v = isZ1 ? 0.09 + noise * 0.18 : 0.55 + noise * 0.2;
      else if (layerId === 'soilmoist') v = cy * 0.5 + noise * 0.2 + (isZ2 ? 0.2 : 0);
      else v = 0.36 + noise * 0.38;
      row.push(Math.max(0, Math.min(1, v)));
    }
    grid.push(row);
  }
  return grid;
}

                  </text>
                  <circle cx={cx} cy={cy} r={4} fill={z.color} opacity={0.9} />
                  {(isH || isSel) && (
                    <text x={cx} y={cy - r - 20} textAnchor="middle"
                      fontSize={9.5} fill={z.color} fontFamily="Inter" fontWeight={600}>
                      {z.label} · {Math.round(z.prospectivity * 100)}% · Score {z.score}
                    </text>
                  )}
                </g>
              );
            })}

            {/* Mineral occurrences */}
            {occLayer && OCCURRENCES.map((o, i) => (
              <polygon key={i}
                points={`${o.x * VW},${o.y * VH - 6} ${o.x * VW - 4},${o.y * VH + 3} ${o.x * VW + 4},${o.y * VH + 3}`}
                fill={o.validated ? '#e91e63' : '#ff8a80'} opacity={0.85} />
            ))}

            {/* Drill holes */}
            {drillLayer && DRILL_HOLES.map((dh, i) => (
              <g key={i}>
                <circle cx={dh.x * VW} cy={dh.y * VH} r={5}
                  fill="none"
                  stroke={dh.status === 'completed' ? '#00bcd4' : '#ffeb3b'}
                  strokeWidth={1.5} opacity={0.8} />
                <circle cx={dh.x * VW} cy={dh.y * VH} r={2.5}
                  fill={dh.status === 'completed' ? '#00bcd4' : '#ffeb3b'} opacity={0.9} />
              </g>
            ))}

            {/* Selected crosshair */}
            {(() => {
              const sx = selectedLoc.x * VW, sy = selectedLoc.y * VH;
              return (
                <g>
                  <line x1={sx - 12} y1={sy} x2={sx + 12} y2={sy} stroke="white" strokeWidth={0.8} opacity={0.5} />
                  <line x1={sx} y1={sy - 12} x2={sx} y2={sy + 12} stroke="white" strokeWidth={0.8} opacity={0.5} />
                  <circle cx={sx} cy={sy} r={5} fill="none" stroke="white" strokeWidth={0.7} opacity={0.5} />
                </g>
              );
            })()}

            {/* Compass */}
            <g transform={`translate(${VW - 26}, 26)`}>
              <circle cx={0} cy={0} r={16} fill="hsl(210 22% 9% / 0.9)" stroke="hsl(210 18% 18%)" strokeWidth={0.8} />
              <polygon points="0,-10 2.5,2 0,6 -2.5,2" fill="hsl(var(--amber))" opacity={0.9} />
              <polygon points="0,10 2.5,-2 0,-6 -2.5,-2" fill="hsl(210 10% 38%)" opacity={0.7} />
              <text x={0} y={-4} textAnchor="middle" fontSize={7} fill="hsl(var(--amber))" fontFamily="Inter" fontWeight={700}>N</text>
            </g>

            {/* Scale */}
            <g transform={`translate(${VW * 0.53}, ${VH - 13})`}>
              <line x1={0} y1={0} x2={55} y2={0} stroke="rgba(255,255,255,0.3)" strokeWidth={1.5} />
              <line x1={0} y1={-3} x2={0} y2={3} stroke="rgba(255,255,255,0.3)" strokeWidth={1} />
              <line x1={55} y1={-3} x2={55} y2={3} stroke="rgba(255,255,255,0.3)" strokeWidth={1} />
              <text x={27} y={-5} textAnchor="middle" fontSize={8} fill="rgba(255,255,255,0.4)">~1 km</text>
            </g>

            {/* Coordinates */}
            <g>
              <rect x={6} y={VH - 17} width={165} height={13} rx={2} fill="hsl(210 22% 9% / 0.85)" />
              <text x={10} y={VH - 7} fontSize={8} fill="hsl(30 8% 60%)" fontFamily="JetBrains Mono, monospace">
                {locLat}°N · {locLng}°E
              </text>
            </g>

            {/* Prospectivity legend */}
            {prospLayer && (
              <g transform={`translate(8, ${VH - 66})`}>
                <rect x={0} y={0} width={118} height={58} rx={2}
                  fill="hsl(210 22% 9% / 0.9)" stroke="hsl(210 18% 18%)" strokeWidth={0.7} />
                <text x={8} y={13} fontSize={7.5} fill="hsl(30 5% 44%)" fontFamily="Inter" fontWeight={600} textAnchor="start">
                  Mn PROSPECTIVITY
                </text>
                {[
                  { y: 24, label: 'High (>75%)',     color: '#c62828' },
                  { y: 36, label: 'Medium (50–75%)', color: '#f57c00' },
                  { y: 48, label: 'Low (<50%)',       color: '#78909c' },
                ].map(item => (
                  <g key={item.label}>
                    <circle cx={14} cy={item.y} r={4} fill={item.color} opacity={0.85} />
                    <text x={24} y={item.y + 3} fontSize={7.5} fill="hsl(30 8% 60%)" fontFamily="Inter">
                      {item.label}
                    </text>
                  </g>
                ))}
              </g>
            )}

            {/* Drill hole legend */}
            {drillLayer && (
              <g transform={`translate(130, ${VH - 42})`}>
                <rect x={0} y={0} width={104} height={34} rx={2}
                  fill="hsl(210 22% 9% / 0.9)" stroke="hsl(210 18% 18%)" strokeWidth={0.7} />
                <circle cx={14} cy={12} r={4} fill="none" stroke="#00bcd4" strokeWidth={1.2} opacity={0.8} />
                <circle cx={14} cy={12} r={2} fill="#00bcd4" opacity={0.8} />
                <text x={24} y={15} fontSize={7.5} fill="hsl(30 8% 60%)" fontFamily="Inter">Completed</text>
                <circle cx={14} cy={26} r={4} fill="none" stroke="#ffeb3b" strokeWidth={1.2} opacity={0.8} />
                <circle cx={14} cy={26} r={2} fill="#ffeb3b" opacity={0.8} />
                <text x={24} y={29} fontSize={7.5} fill="hsl(30 8% 60%)" fontFamily="Inter">In-progress</text>
              </g>
            )}
          </svg>

          {/* Raster ramp legend */}
          {primaryRasterLayer && RAMPS[primaryRasterLayer.id] && (
            <div
              className="absolute bottom-14 right-3 rounded-sm px-3 py-2.5 animate-fade-in"
              style={{
                background: 'hsl(210 22% 9% / 0.9)',
                border: '1px solid hsl(210 18% 18%)',
              }}
            >
              <div className="section-label mb-1.5">{primaryRasterLayer.label}</div>
              <div className="w-28 h-2 rounded-sm" style={{
                background: `linear-gradient(to right, ${RAMPS[primaryRasterLayer.id].join(', ')})`,
              }} />
              <div className="flex justify-between mt-0.5">
                <span className="text-[9px] font-mono" style={{ color: 'hsl(var(--text-muted))' }}>Low</span>
                <span className="text-[9px] font-mono" style={{ color: 'hsl(var(--text-muted))' }}>High</span>
              </div>
            </div>
          )}
        </div>
      </div>

      {/* ── RIGHT: Analysis panel ─────────────────────────────── */}
      <div
        className="w-[280px] shrink-0 flex flex-col overflow-hidden"
        style={{
          background: 'hsl(var(--nav-bg) / 0.96)',
          borderLeft: '1px solid hsl(var(--nav-border))',
        }}
      >
        {/* Tabs */}
        <div className="flex shrink-0" style={{ borderBottom: '1px solid hsl(var(--nav-border))' }}>
          {[
            { id: 'zone'     as RightTab, label: 'Zone Detail' },
            { id: 'targets'  as RightTab, label: 'Targets' },
            { id: 'spectral' as RightTab, label: 'Spectral' },
          ].map(tab => (
            <button
              key={tab.id}
              onClick={() => setRightTab(tab.id)}
              className="flex-1 py-2.5 text-[11px] font-medium transition-colors"
              style={{
                color: rightTab === tab.id ? 'hsl(var(--text-strong))' : 'hsl(var(--text-muted))',
                borderBottom: rightTab === tab.id ? '2px solid hsl(var(--amber))' : '2px solid transparent',
              }}
            >
              {tab.label}
            </button>
          ))}
        </div>

        <div className="flex-1 overflow-y-auto">

          {/* ── Zone Detail ── */}
          {rightTab === 'zone' && (
            <div className="p-4 space-y-4">
              {/* Zone header */}
              <div>
                <div className="section-label mb-2">Exploration Target #{t.rank}</div>
                <div className="flex items-center justify-between gap-2 mb-3">
                  <h2 className="text-[15px] font-bold" style={{ color: 'hsl(var(--text-strong))' }}>{t.label}</h2>
                  <span className={t.type === 'high' ? 'prospec-high' : t.type === 'medium' ? 'prospec-medium' : 'prospec-low'}>
                    {t.type}
                  </span>
                </div>

                {/* Score + prospectivity */}
                <div className="grid grid-cols-2 gap-3 mb-3">
                  <div className="rounded-sm p-3 text-center"
                    style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--border))' }}>
                    <div className="section-label mb-1">Priority Score</div>
                    <div className="text-[26px] font-bold tabular-nums"
                      style={{ color: t.type === 'high' ? 'hsl(var(--red-light))' : 'hsl(var(--amber-light))', letterSpacing: '-0.02em' }}>
                      {t.score}
                    </div>
                    <div className="text-[10px]" style={{ color: 'hsl(var(--text-muted))' }}>/100</div>
                  </div>
                  <div className="rounded-sm p-3 text-center"
                    style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--border))' }}>
                    <div className="section-label mb-1">Confidence</div>
                    <div className="text-[26px] font-bold tabular-nums"
                      style={{ color: 'hsl(var(--text-strong))', letterSpacing: '-0.02em' }}>
                      {Math.round(t.confidence * 100)}%
                    </div>
                    <div className="text-[10px]" style={{ color: uncertaintyColor }}>
                      {t.uncertainty} uncertainty
                    </div>
                  </div>
                </div>

                {/* Prospectivity bar */}
                <div className="space-y-1.5">
                  <div className="flex justify-between items-center">
                    <span className="text-[12px]" style={{ color: 'hsl(var(--text-secondary))' }}>Model Prospectivity</span>
                    <span className="text-[15px] font-bold tabular-nums"
                      style={{ color: t.type === 'high' ? 'hsl(var(--red-light))' : 'hsl(var(--amber-light))' }}>
                      {Math.round(t.prospectivity * 100)}%
                    </span>
                  </div>
                  <div className="progress-track" style={{ height: '5px' }}>
                    <div className="progress-fill" style={{
                      width: `${t.prospectivity * 100}%`,
                      background: t.type === 'high' ? 'hsl(var(--red))' : 'hsl(var(--amber))',
                    }} />
                  </div>
                  <div className="text-[10px]" style={{ color: 'hsl(var(--text-muted))' }}>
                    XGBoost v1.7 · {t.drillStatus}
                  </div>
                </div>
              </div>

              {/* Why this prediction */}
              <div>
                <div className="section-label mb-2">Why This Prediction?</div>
                {t.reasons.map((r, i) => (
                  <div key={i} className="factor-row">
                    <span className="text-[9px] px-1.5 py-0.5 rounded-sm shrink-0 font-semibold"
                      style={{ background: 'hsl(var(--surface-select))', color: 'hsl(var(--taupe))' }}>
                      {r.tag}
                    </span>
                    <span className="flex-1 text-[11px]" style={{ color: 'hsl(var(--text-secondary))' }}>{r.label}</span>
                    <div className="w-10 shrink-0">
                      <div className="h-1 rounded-sm" style={{ background: 'hsl(var(--surface-select))' }}>
                        <div className="h-full rounded-sm" style={{
                          width: `${r.strength * 100}%`,
                          background: 'hsl(var(--amber))',
                        }} />
                      </div>
                    </div>
                  </div>
                ))}
              </div>

              {/* Validation */}
              <div>
                <div className="section-label mb-2">Validation Status</div>
                {t.validation.map(v => (
                  <div key={v.label} className="flex items-center gap-2.5 py-1.5 factor-row">
                    <div className="w-4 h-4 rounded-sm flex items-center justify-center shrink-0"
                      style={{ background: v.ok ? 'hsl(var(--green-dim))' : 'hsl(var(--surface-select))' }}>
                      {v.ok
                        ? <CheckCircle2 className="w-3 h-3" style={{ color: 'hsl(var(--green-light))' }} />
                        : <span style={{ fontSize: 9, color: 'hsl(var(--text-muted))' }}>—</span>}
                    </div>
                    <span className="text-[11px]" style={{ color: v.ok ? 'hsl(var(--text-secondary))' : 'hsl(var(--text-muted))' }}>
                      {v.label}
                    </span>
                  </div>
                ))}
              </div>

              {/* Recommendation */}
              <div
                className="rounded-sm p-3"
                style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--border))' }}
              >
                <div className="section-label mb-1.5">Recommended Next Step</div>
                <div className="text-[12px] font-semibold mb-1" style={{ color: 'hsl(var(--amber-light))' }}>
                  {t.recommendedStep}
                </div>
                <p className="text-[11px] leading-relaxed" style={{ color: 'hsl(var(--text-secondary))' }}>
                  {t.recommendation}
                </p>
              </div>

              {/* Disclaimer */}
              <div className="rounded-sm p-3" style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--border))' }}>
                <div className="flex items-start gap-1.5">
                  <Info className="w-3 h-3 shrink-0 mt-0.5" style={{ color: 'hsl(var(--taupe))' }} />
                  <p className="text-[10px] leading-relaxed" style={{ color: 'hsl(var(--text-muted))' }}>
                    Probabilistic exploration target. Not a confirmed Mn reserve. Drilling confirmation required before resource classification.
                  </p>
                </div>
              </div>
            </div>
          )}

          {/* ── Exploration Targets list ── */}
          {rightTab === 'targets' && (
            <div className="p-4 space-y-3">
              <div>
                <div className="section-label">Exploration Target Ranking</div>
                <p className="text-[11px] mt-1 leading-relaxed" style={{ color: 'hsl(var(--text-muted))' }}>
                  Ranked by combined priority score. Click to view evidence and explanation.
                </p>
              </div>

              {EXPLORATION_TARGETS.map(tgt => (
                <div
                  key={tgt.id}
                  className={`target-row ${selectedTarget?.id === tgt.id ? 'target-row-active' : ''}`}
                  onClick={() => { setSelectedTarget(tgt); setSelectedLoc({ x: tgt.cx, y: tgt.cy }); setRightTab('zone'); }}
                >
                  <div
                    className="w-6 h-6 rounded-sm flex items-center justify-center shrink-0 text-[11px] font-bold"
                    style={{ background: tgt.type === 'high' ? tgt.color : 'hsl(var(--surface-select))', color: tgt.type === 'high' ? 'white' : 'hsl(var(--text-muted))' }}
                  >
                    {tgt.rank}
                  </div>
                  <div className="flex-1 min-w-0">
                    <div className="text-[13px] font-semibold" style={{ color: 'hsl(var(--text-primary))' }}>{tgt.label}</div>
                    <div className="flex items-center gap-2 mt-0.5 text-[11px]" style={{ color: 'hsl(var(--text-muted))' }}>
                      <span>{Math.round(tgt.prospectivity * 100)}% prosp.</span>
                      <span style={{ color: tgt.uncertainty === 'low' ? 'hsl(var(--green-light))' : tgt.uncertainty === 'medium' ? 'hsl(var(--amber-light))' : 'hsl(var(--red-light))' }}>
                        {tgt.uncertainty} uncert.
                      </span>
                    </div>
                  </div>
                  <div className="text-right shrink-0">
                    <div className="text-[18px] font-bold tabular-nums"
                      style={{ color: tgt.type === 'high' ? 'hsl(var(--red-light))' : tgt.type === 'medium' ? 'hsl(var(--amber-light))' : 'hsl(var(--text-muted))' }}>
                      {tgt.score}
                    </div>
                    <div className="text-[9px]" style={{ color: 'hsl(var(--text-muted))' }}>/100</div>
                  </div>
                </div>
              ))}

              <div className="rounded-sm p-3" style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--amber-dim))' }}>
                <div className="section-label mb-1.5">Workflow</div>
                {[
                  'Satellite + Geological Evidence',
                  'Prospectivity Prediction',
                  'Explainability Review',
                  'Target Ranking',
                  'Field / Drilling Priority',
                ].map((step, i) => (
                  <div key={step} className="flex items-center gap-2 py-1">
                    <div className="w-4 h-4 rounded-sm flex items-center justify-center shrink-0 text-[9px] font-bold"
                      style={{ background: 'hsl(var(--amber-dim) / 0.3)', color: 'hsl(var(--amber-light))' }}>
                      {i + 1}
                    </div>
                    <span className="text-[11px]" style={{ color: 'hsl(var(--text-secondary))' }}>{step}</span>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* ── Spectral ── */}
          {rightTab === 'spectral' && (
            <div className="p-4 space-y-4">
              <div>
                <div className="section-label mb-1">Spectral Profile</div>
                <div className="text-[11px]" style={{ color: 'hsl(var(--text-muted))' }}>
                  Selected point · Sentinel-2 L2A
                </div>
              </div>
              <ResponsiveContainer width="100%" height={140}>
                <LineChart data={SPECTRAL_DATA} margin={{ top: 4, right: 4, left: -20, bottom: 0 }}>
                  <XAxis dataKey="band" tick={{ fontSize: 8 }} axisLine={false} tickLine={false} />
                  <YAxis tick={{ fontSize: 8 }} axisLine={false} tickLine={false} domain={[0, 0.4]} />
                  <Tooltip
                    contentStyle={{ background: 'hsl(var(--nav-bg))', border: '1px solid hsl(var(--border))', borderRadius: 3, fontSize: 11 }}
                    formatter={(v: number) => [(v as number).toFixed(3), 'Reflectance']}
                  />
                  <Line type="monotone" dataKey="v" stroke="hsl(var(--amber-light))" strokeWidth={1.5}
                    dot={{ r: 3, fill: 'hsl(var(--amber-light))' }} name="Reflectance" />
                </LineChart>
              </ResponsiveContainer>

              <div>
                <div className="section-label mb-2">Spectral Indices</div>
                {[
                  { name: 'NDVI',       val: '−0.12', note: 'Bare/disturbed soil' },
                  { name: 'BSI',        val: '0.38',  note: 'Elevated bare soil' },
                  { name: 'FeMg Ratio', val: '1.24',  note: 'Fe/Mg anomaly' },
                  { name: 'Clay Index', val: '0.64',  note: 'Weathered laterite' },
                ].map(idx => (
                  <div key={idx.name} className="factor-row">
                    <span className="text-[11px] font-mono w-18 shrink-0" style={{ color: 'hsl(var(--amber-light))' }}>{idx.name}</span>
                    <span className="text-[12px] w-10 tabular-nums shrink-0" style={{ color: 'hsl(var(--text-primary))' }}>{idx.val}</span>
                    <span className="text-[10px]" style={{ color: 'hsl(var(--text-muted))' }}>{idx.note}</span>
                  </div>
                ))}
              </div>

              <div className="rounded-sm p-3" style={{ background: 'hsl(var(--surface-raised))', border: '1px solid hsl(var(--border))' }}>
                <div className="section-label mb-1.5">Acquisition</div>
                {[
                  ['Sensor', 'Sentinel-2 MSI'],
                  ['Resolution', '10 – 20 m'],
                  ['Date', '2026-08-22'],
                  ['CRS', 'WGS84 / UTM 44N'],
                ].map(([k, v]) => (
                  <div key={k} className="flex justify-between py-1 text-[11px]">
                    <span style={{ color: 'hsl(var(--text-muted))' }}>{k}</span>
                    <span style={{ color: 'hsl(var(--text-secondary))' }}>{v}</span>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
